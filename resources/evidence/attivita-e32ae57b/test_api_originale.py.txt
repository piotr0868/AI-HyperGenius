import unittest, subprocess, json, os, time, socket

class APITest(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.db='test_activities.db'
        # start server
        cls.proc=subprocess.Popen(['python3','main.py','--port','8001','--db',cls.db], stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        time.sleep(1)
    @classmethod
    def tearDownClass(cls):
        cls.proc.terminate()
        cls.proc.wait()
        if os.path.exists(cls.db): os.remove(cls.db)
    def test_get_empty(self):
        import urllib.request
        with urllib.request.urlopen('http://localhost:8001/activities') as r:
            data=json.loads(r.read())
            self.assertEqual(data,[])
    def test_post_create(self):
        import urllib.request, urllib.error
        req=urllib.request.Request('http://localhost:8001/activities',data=json.dumps({'title':'test'}).encode(),headers={'Content-Type':'application/json','Idempotency-Key':'key1'},method='POST')
        with urllib.request.urlopen(req) as r:
            self.assertEqual(r.status,201)
            body=json.loads(r.read())
            self.assertIn('id',body)
            self.assertEqual(body['title'],'test')
    def test_post_duplicate_same(self):
        import urllib.request
        req=urllib.request.Request('http://localhost:8001/activities',data=json.dumps({'title':'test'}).encode(),headers={'Content-Type':'application/json','Idempotency-Key':'key1'},method='POST')
        with urllib.request.urlopen(req) as r:
            self.assertEqual(r.status,200)
    def test_post_duplicate_diff(self):
        import urllib.request, urllib.error
        req=urllib.request.Request('http://localhost:8001/activities',data=json.dumps({'title':'other'}).encode(),headers={'Content-Type':'application/json','Idempotency-Key':'key1'},method='POST')
        with self.assertRaises(urllib.error.HTTPError) as cm:
            urllib.request.urlopen(req)
        self.assertEqual(cm.exception.code,409)
    def test_title_validation(self):
        import urllib.request, urllib.error
        req=urllib.request.Request('http://localhost:8001/activities',data=json.dumps({'title':''}).encode(),headers={'Content-Type':'application/json','Idempotency-Key':'k2'},method='POST')
        with self.assertRaises(urllib.error.HTTPError) as cm:
            urllib.request.urlopen(req)
        self.assertEqual(cm.exception.code,422)

if __name__=='__main__':
    unittest.main()
